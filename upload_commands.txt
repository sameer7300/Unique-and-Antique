# Hostinger Deployment Commands
# Run these commands in your SSH session

# 1. Navigate to the backend directory
cd domains/unique-antique.com/public_html/backend/

# 2. Create virtual environment
python3 -m venv venv
source venv/bin/activate

# 3. Install dependencies
pip install -r requirements.txt

# 4. Set up database (make sure to create PostgreSQL database in hPanel first)
python manage.py makemigrations
python manage.py migrate

# 5. Run production setup script
python setup_production.py

# 6. Test the application
python manage.py runserver 0.0.0.0:8000

# 7. Create a simple WSGI test
cat > test_wsgi.py << 'EOF'
import os
import sys
from pathlib import Path

# Add backend to path
backend_path = Path(__file__).parent
sys.path.insert(0, str(backend_path))

# Set Django settings
os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'config.settings.production')

try:
    from django.core.wsgi import get_wsgi_application
    application = get_wsgi_application()
    print("✅ WSGI application loaded successfully!")
except Exception as e:
    print(f"❌ Error loading WSGI: {e}")
EOF

python test_wsgi.py

# 8. Check if all is working
python manage.py check --deploy
